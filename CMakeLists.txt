cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(USBTrace VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio C++ on Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument user mode" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(USBTrace SDK/USBTrace.cpp)
target_link_libraries(USBTrace PUBLIC winhttp crypt32)
add_library(USBTraceSetupCore Installer/Setup.cpp Installer/DeviceFilter.cpp)
target_link_libraries(USBTraceSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32 cfgmgr32)
add_executable(USBTraceStudio WIN32 App/Main.cpp)
target_link_libraries(USBTraceStudio PRIVATE USBTrace USBTraceSetupCore user32 gdi32 comctl32 shell32)
target_link_options(USBTraceStudio PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(USBTraceCLI App/Console.cpp)
target_link_libraries(USBTraceCLI PRIVATE USBTrace)
add_executable(USBTraceExample Examples/observe.cpp)
target_link_libraries(USBTraceExample PRIVATE USBTrace)
add_executable(USBTraceTests Tests/Protocol.cpp)
target_link_libraries(USBTraceTests PRIVATE USBTraceSetupCore)
enable_testing()
add_test(NAME ProtocolAndInstallerRefusal COMMAND USBTraceTests)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft signed driver package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanUSBTrace.inf TitanUSBTrace.cat TitanUSBTrace.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete signed package")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanUSBTrace.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanUSBTrace.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanUSBTrace.sys\"\n")
endif()
add_executable(USBTraceSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(USBTraceSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target USBTraceSetup USBTraceSetupConsole)
 target_link_libraries(${target} PRIVATE USBTraceSetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
foreach(target USBTraceStudio USBTraceCLI USBTraceExample USBTraceTests USBTraceSetup USBTraceSetupConsole)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
