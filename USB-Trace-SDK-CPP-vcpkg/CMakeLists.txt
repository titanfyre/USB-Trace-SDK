cmake_minimum_required(VERSION 3.24)
project(USBTraceSDK VERSION 0.1.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This SDK requires Windows x64. No driver is installed by this build.")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(TitanUSBTraceSDK STATIC
  "sdk/SDK/USBTrace.cpp"
)
add_library(USBTraceSDK::sdk ALIAS TitanUSBTraceSDK)
set_target_properties(TitanUSBTraceSDK PROPERTIES EXPORT_NAME sdk)
target_compile_features(TitanUSBTraceSDK PUBLIC cxx_std_17)
target_compile_definitions(TitanUSBTraceSDK PUBLIC )
target_compile_definitions(TitanUSBTraceSDK PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(TitanUSBTraceSDK PRIVATE /utf-8 /EHsc /W4)
endif()
target_include_directories(TitanUSBTraceSDK PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/sdk>"
  "$<INSTALL_INTERFACE:include/>"
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/sdk/SDK>"
  "$<INSTALL_INTERFACE:include/SDK>"
)
target_link_libraries(TitanUSBTraceSDK PUBLIC advapi32 cfgmgr32 comctl32 crypt32 gdi32 newdev ole32 setupapi shell32 user32 winhttp wintrust)
install(TARGETS TitanUSBTraceSDK EXPORT USBTraceSDKTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY sdk/ DESTINATION include FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp" PATTERN "*.inl")
install(FILES LICENSE DESTINATION share/usb-trace-sdk)
install(EXPORT USBTraceSDKTargets NAMESPACE USBTraceSDK:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/USBTraceSDK)
configure_file("USBTraceSDKConfig.cmake.in" "${CMAKE_CURRENT_BINARY_DIR}/USBTraceSDKConfig.cmake" @ONLY)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/USBTraceSDKConfigVersion.cmake" VERSION 0.1.1 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/USBTraceSDKConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/USBTraceSDKConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/USBTraceSDK)
